        }
        log.Println("Conexión exitosa a PostgreSQL Primario (Puerto 5432)")

        // 3. Conexión a PostgreSQL Réplica (Puerto 5433 - Analítica Pesada P4 / ADR-17)
        replicaConnStr := "postgres://postgres:password@localhost:5433/trotamundos?sslmode=disable"
        replicaPool, err := pgxpool.New(ctx, replicaConnStr)
        if err != nil {
                log.Printf("Advertencia: No se pudo conectar a la Réplica PostgreSQL (Puerto 5433): %v", err)
        } else {
                if err := replicaPool.Ping(ctx); err != nil {
                        log.Printf("Advertencia: Réplica no responde en el puerto 5433, fallback activo.")
                } else {
                        log.Println("Conexión exitosa a PostgreSQL Réplica de Analítica (Puerto 5433)")
                }
                defer replicaPool.Close()
        }

        // 4. Repositorios
        taskRepo := repository.NewPostgresTaskRepository(primaryPool)
        var tripRepo domain.TripRepository = nil
        var userRepo domain.UserRepository = nil
        var shareRepo domain.ShareRepository = nil

        // Usar réplica si está disponible; de lo contrario, conectar al primario como fallback
        var analyticsRepo domain.AnalyticsRepository
        if replicaPool != nil {
                analyticsRepo = repository.NewPostgresAnalyticsRepository(replicaPool)
        } else {
                analyticsRepo = repository.NewPostgresAnalyticsRepository(primaryPool)
        }

        // 5. Servicios y Casos de Uso
        encryptor := &DummyEncryptor{}

        tripUC := usecase.NewTripUsecase(tripRepo)
        emailUC := usecase.NewEmailUsecase(userRepo, encryptor)
        shareUC := usecase.NewShareUsecase(shareRepo)
        _ = usecase.NewAnalyticsUsecase(analyticsRepo)

        // 6. Configurar e Iniciar Background Worker
        taskWorker := worker.NewTaskWorker(taskRepo, 3*time.Second, 3)

        taskWorker.RegisterHandler("EMAIL_SCAN", func(ctx context.Context, payload []byte) error {
                log.Printf("[Handler EMAIL_SCAN] Escaneando correo asíncronamente: %s", string(payload))
                return nil
        })

        taskWorker.RegisterHandler("FETCH_PROVIDER_UPDATE", func(ctx context.Context, payload []byte) error {
                log.Printf("[Handler FETCH_PROVIDER] Actualizando datos de proveedor: %s", string(payload))
                return nil
        })

        taskWorker.Start(ctx)

        // 7. Servidor HTTP
        mux := http.NewServeMux()
        handler := delivery.NewHTTPHandler(tripUC, emailUC, shareUC)
        handler.RegisterRoutes(mux)

        server := &http.Server{
                Addr:    ":8080",
                Handler: mux,
        }

        go func() {
                log.Println("Servidor Trotamundos API ejecutándose en puerto :8080...")
                if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
                        log.Fatalf("Error en el servidor HTTP: %v", err)
                }
        }()

        // 8. Graceful Shutdown
        stop := make(chan os.Signal, 1)
        signal.Notify(stop, os.Interrupt, syscall.SIGTERM)

        <-stop
        log.Println("\nIniciando apagado controlado del sistema (Graceful Shutdown)...")

        taskWorker.Stop()

        shutdownCtx, shutdownCancel := context.WithTimeout(context.Background(), 5*time.Second)
        defer shutdownCancel()

        if err := server.Shutdown(shutdownCtx); err != nil {
                log.Fatalf("Error al cerrar el servidor HTTP: %v", err)
        }

        log.Println("Sistema Trotamundos detenido correctamente.")
}

package i18n

import (
        "encoding/json"
        "fmt"
        "os"
        "path/filepath"
        "sync"
)

type Translator struct {
        mu           sync.RWMutex
        translations map[string]map[string]string
        defaultLang  string
}

func NewTranslator(configDir string, defaultLang string) (*Translator, error) {
        t := &Translator{
                translations: make(map[string]map[string]string),
                defaultLang:  defaultLang,
        }

        files, err := filepath.Glob(filepath.Join(configDir, "*.json"))
        if err != nil {
                return nil, fmt.Errorf("error al buscar archivos i18n: %w", err)
        }

        for _, file := range files {
                lang := filepath.Base(file)
                lang = lang[:len(lang)-len(filepath.Ext(lang))] // Extrae 'es', 'en', etc.

                data, err := os.ReadFile(file)
                if err != nil {
                        return nil, fmt.Errorf("error al leer archivo %s: %w", file, err)
                }

                var langMap map[string]string
                if err := json.Unmarshal(data, &langMap); err != nil {
                        return nil, fmt.Errorf("error al parsear JSON %s: %w", file, err)
                }

                t.translations[lang] = langMap
        }

        return t, nil
}

func (t *Translator) Translate(lang string, key string) string {
        t.mu.RLock()
        defer t.mu.RUnlock()

        if langMap, exists := t.translations[lang]; exists {
                if val, ok := langMap[key]; ok {
                        return val
                }
        }

        // Fallback al idioma por defecto o devolver la clave recibida
        if defaultMap, exists := t.translations[t.defaultLang]; exists {
                if val, ok := defaultMap[key]; ok {
                        return val
                }
        }

        return key
}

// Interfaces mínimas que necesita la capa de entrega. Los casos de uso
// concretos (*usecase.TripUsecase, etc.) las satisfacen sin cambios.
type TripService interface {
        GetItinerarySummary(ctx context.Context, tripID string) ([]byte, error)
        UpdateItinerarySummary(ctx context.Context, tripID string, summary []byte) error
}

type EmailService interface {
        LinkEmailAccount(ctx context.Context, userID string, rawAccessToken string) error
        RevokeEmailAccount(ctx context.Context, userID string) error
}

type ShareService interface {
        CreateItineraryShareLink(ctx context.Context, tripID string, duration time.Duration) (string, error)
        AccessSharedItinerary(ctx context.Context, token string) (string, error)
        RevokeLink(ctx context.Context, token string) error
}

type HTTPHandler struct {
        trip  TripService
        email EmailService
        share ShareService
}

func NewHTTPHandler(trip TripService, email EmailService, share ShareService) *HTTPHandler {
        return &HTTPHandler{trip: trip, email: email, share: share}
}

// RegisterRoutes usa los patrones con método y {wildcards} de Go 1.22+.
func (h *HTTPHandler) RegisterRoutes(mux *http.ServeMux) {
        mux.HandleFunc("GET /api/trips/{id}/summary", h.getSummary)
        mux.HandleFunc("PUT /api/trips/{id}/summary", h.updateSummary)
        mux.HandleFunc("POST /api/trips/{id}/share", h.createShare)

        mux.HandleFunc("POST /api/users/{id}/email", h.linkEmail)
        mux.HandleFunc("DELETE /api/users/{id}/email", h.revokeEmail)

        mux.HandleFunc("GET /api/shared/{token}", h.accessShared)
        mux.HandleFunc("DELETE /api/shared/{token}", h.revokeShare)
}

// ---------------------------------------------------------------- helpers

func writeJSON(w http.ResponseWriter, status int, body any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(body)
}

func writeError(w http.ResponseWriter, status int, msg string) {
        writeJSON(w, status, map[string]string{"error": msg})
}

func serviceUnavailable(w http.ResponseWriter) {
        writeError(w, http.StatusServiceUnavailable, "servicio no disponible")
}

func decodeBody(w http.ResponseWriter, r *http.Request, dst any) bool {
        r.Body = http.MaxBytesReader(w, r.Body, 1<<20) // 1 MB
        if err := json.NewDecoder(r.Body).Decode(dst); err != nil {
                writeError(w, http.StatusBadRequest, "cuerpo JSON inválido")
                return false
        }
        return true
}

func isNotFound(err error) bool {
        msg := err.Error()
        return strings.Contains(msg, "no encontrado") || strings.Contains(msg, "inválido") || strings.Contains(msg, "expirado")
}

// ---------------------------------------------------------------- trips

func (h *HTTPHandler) getSummary(w http.ResponseWriter, r *http.Request) {
        if h.trip == nil {
                serviceUnavailable(w)
                return
        }
        summary, err := h.trip.GetItinerarySummary(r.Context(), r.PathValue("id"))
        if err != nil {
                if isNotFound(err) {
                        writeError(w, http.StatusNotFound, "viaje no encontrado")
                        return
                }
                writeError(w, http.StatusInternalServerError, "error interno")
                return
        }
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(http.StatusOK)
        _, _ = w.Write(summary) // JSONB crudo, sin re-serializar
}

func (h *HTTPHandler) updateSummary(w http.ResponseWriter, r *http.Request) {
        if h.trip == nil {
                serviceUnavailable(w)
                return
        }
        var raw json.RawMessage
        if !decodeBody(w, r, &raw) {
                return
        }
        if err := h.trip.UpdateItinerarySummary(r.Context(), r.PathValue("id"), raw); err != nil {
                if isNotFound(err) {
                        writeError(w, http.StatusNotFound, "viaje no encontrado")
                        return
                }
                writeError(w, http.StatusInternalServerError, "error interno")
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

// ---------------------------------------------------------------- email (SEC1)

func (h *HTTPHandler) linkEmail(w http.ResponseWriter, r *http.Request) {
        if h.email == nil {
                serviceUnavailable(w)
                return
        }
        var body struct {
                AccessToken string `json:"access_token"`
        }
        if !decodeBody(w, r, &body) {
                return
        }
        if body.AccessToken == "" {
                writeError(w, http.StatusBadRequest, "access_token es requerido")
                return
        }
        if err := h.email.LinkEmailAccount(r.Context(), r.PathValue("id"), body.AccessToken); err != nil {
                writeError(w, http.StatusInternalServerError, "error interno")
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

func (h *HTTPHandler) revokeEmail(w http.ResponseWriter, r *http.Request) {
        if h.email == nil {
                serviceUnavailable(w)
                return
        }
        if err := h.email.RevokeEmailAccount(r.Context(), r.PathValue("id")); err != nil {
                writeError(w, http.StatusInternalServerError, "error interno")
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

// ---------------------------------------------------------------- share (SEC2)

func (h *HTTPHandler) createShare(w http.ResponseWriter, r *http.Request) {
        if h.share == nil {
                serviceUnavailable(w)
                return
        }
        var body struct {
                DurationHours int `json:"duration_hours"`
        }
        if !decodeBody(w, r, &body) {
                return
        }
        if body.DurationHours <= 0 {
                writeError(w, http.StatusBadRequest, "duration_hours debe ser mayor a 0")
                return
        }
        token, err := h.share.CreateItineraryShareLink(r.Context(), r.PathValue("id"), time.Duration(body.DurationHours)*time.Hour)
        if err != nil {
                writeError(w, http.StatusInternalServerError, "error interno")
                return
        }
        writeJSON(w, http.StatusCreated, map[string]string{"token": token})
}

func (h *HTTPHandler) accessShared(w http.ResponseWriter, r *http.Request) {
        if h.share == nil {
                serviceUnavailable(w)
                return
        }
        tripID, err := h.share.AccessSharedItinerary(r.Context(), r.PathValue("token"))
        if err != nil || tripID == "" {
                writeError(w, http.StatusNotFound, "enlace inválido o expirado")
                return
        }
        writeJSON(w, http.StatusOK, map[string]string{"trip_id": tripID})
}

func (h *HTTPHandler) revokeShare(w http.ResponseWriter, r *http.Request) {
        if h.share == nil {
                serviceUnavailable(w)
                return
        }
        if err := h.share.RevokeLink(r.Context(), r.PathValue("token")); err != nil {
                writeError(w, http.StatusInternalServerError, "error interno")
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

type DBTX interface {
        Exec(ctx context.Context, sql string, arguments ...any) (pgconn.CommandTag, error)
        QueryRow(ctx context.Context, sql string, args ...any) pgx.Row
        Begin(ctx context.Context) (pgx.Tx, error)
}

type DB struct {
        Pool *pgxpool.Pool
}

func NewDB(ctx context.Context, connString string) (*DB, error) {
        poolConfig, err := pgxpool.ParseConfig(connString)
        if err != nil {
                return nil, fmt.Errorf("error al parsear config de pgx: %w", err)
        }

        pool, err := pgxpool.NewWithConfig(ctx, poolConfig)
        if err != nil {
                return nil, fmt.Errorf("error al conectar a la base de datos: %w", err)
        }

        if err := pool.Ping(ctx); err != nil {
                return nil, fmt.Errorf("error haciendo ping a la base de datos: %w", err)
        }

        return &DB{Pool: pool}, nil
}

func (r *PostgresTaskRepository) ClaimNextPending(ctx context.Context) (*domain.Task, error) {
        tx, err := r.db.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("error al iniciar transacción para tarea: %w", err)
        }
        defer tx.Rollback(ctx)

        query := `
                SELECT id, type, payload, status, attempts, created_at
                FROM tasks
                WHERE status = 'PENDING'
                ORDER BY created_at ASC
                LIMIT 1
                FOR UPDATE SKIP LOCKED;`

        var task domain.Task
        var statusStr string

        err = tx.QueryRow(ctx, query).Scan(
                &task.ID,
                &task.Type,
                &task.Payload,
                &statusStr,
                &task.Attempts,
                &task.CreatedAt,
        )

        if err != nil {
                if errors.Is(err, pgx.ErrNoRows) {
                        return nil, nil
                }
                return nil, fmt.Errorf("error al reclamar tarea pendiente: %w", err)
        }

        task.Status = domain.TaskStatus(statusStr)

        updateQuery := `
                UPDATE tasks 
                SET status = 'PROCESSING', attempts = attempts + 1 
                WHERE id = $1;`

        _, err = tx.Exec(ctx, updateQuery, task.ID)
        if err != nil {
                return nil, fmt.Errorf("error al actualizar estado de tarea a PROCESSING: %w", err)
        }

        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("error al confirmar transacción de tarea: %w", err)
        }

        task.Status = domain.TaskProcessing
        task.Attempts++
        return &task, nil
}

func (r *PostgresTaskRepository) MarkDone(ctx context.Context, id string) error {
        query := `UPDATE tasks SET status = 'DONE' WHERE id = $1;`
        _, err := r.db.Exec(ctx, query, id)
        if err != nil {
                return fmt.Errorf("error al marcar tarea como DONE: %w", err)
        }
        return nil
}

func (r *PostgresTaskRepository) MarkFailed(ctx context.Context, id string, reason string) error {
        query := `UPDATE tasks SET status = 'FAILED' WHERE id = $1;`
        _, err := r.db.Exec(ctx, query, id)
        if err != nil {
                return fmt.Errorf("error al marcar tarea como FAILED: %w", err)
        }
        return nil
}

func (r *PostgresTaskRepository) MarkRetry(ctx context.Context, id string) error {
        query := `UPDATE tasks SET status = 'PENDING' WHERE id = $1;`
        _, err := r.db.Exec(ctx, query, id)
        if err != nil {
                return fmt.Errorf("error al devolver tarea %s a PENDING para reintento: %w", id, err)
        }
        return nil
}

        db DBTX
}

func NewPostgresTripRepository(db DBTX) *PostgresTripRepository {
        return &PostgresTripRepository{db: db}
}

func (r *PostgresTripRepository) GetSummaryByID(ctx context.Context, id string) ([]byte, error) {
        query := `SELECT itinerary_summary FROM trips WHERE id = $1`

        var summary []byte
        err := r.db.QueryRow(ctx, query, id).Scan(&summary)
        if err != nil {
                if errors.Is(err, pgx.ErrNoRows) {
                        return nil, fmt.Errorf("viaje no encontrado: %w", err)
                }
                return nil, fmt.Errorf("error al consultar el resumen del viaje: %w", err)
        }

        return summary, nil
}

func (r *PostgresTripRepository) UpdateSummary(ctx context.Context, id string, summary []byte, currentVersion int) error {
        query := `
                UPDATE trips 
                SET itinerary_summary = $1::jsonb, 
                        version = version + 1, 
                        updated_at = NOW() 
                WHERE id = $2 AND version = $3`

        result, err := r.db.Exec(ctx, query, string(summary), id, currentVersion)
        if err != nil {
                return fmt.Errorf("error al actualizar el resumen del viaje: %w", err)
        }

        // Si RowsAffected() es 0, significa que otro proceso modificó el viaje 
        // al mismo tiempo (colisión de versión) o el ID no existe.
        if result.RowsAffected() == 0 {
                return fmt.Errorf("conflicto de concurrencia o viaje no encontrado para ID %s (versión %d)", id, currentVersion)
        }

        return nil
}

// Genera un enlace seguro y único para compartir un itinerario
func (uc *ShareUsecase) CreateItineraryShareLink(ctx context.Context, tripID string, duration time.Duration) (string, error) {
        if tripID == "" {
                return "", errors.New("el ID de viaje no puede estar vacío")
        }

        bytes := make([]byte, 16)
        if _, err := rand.Read(bytes); err != nil {
                return "", fmt.Errorf("error al generar token aleatorio: %w", err)
        }
        token := hex.EncodeToString(bytes)
        expiresAt := time.Now().Add(duration)

        err := uc.shareRepo.CreateShareLink(ctx, token, tripID, expiresAt)
        if err != nil {
                return "", fmt.Errorf("error al guardar enlace compartido: %w", err)
        }

        return token, nil
}

package worker

import (
        "context"
        "errors"
        "fmt"
        "log"
        "sync"
        "time"

        "github.com/lt-tunesi/TpFinalIng/internal/domain"
)

// TaskHandler define la función responsable de procesar la lógica de un tipo de tarea
type TaskHandler func(ctx context.Context, payload []byte) error

// TaskWorker gestiona el procesamiento asíncrono en segundo plano
type TaskWorker struct {
        taskRepo    domain.TaskRepository
        handlers    map[string]TaskHandler
        interval    time.Duration
        maxAttempts int
        stopChan    chan struct{}
        wg          sync.WaitGroup
}

// NewTaskWorker crea una nueva instancia del worker
func NewTaskWorker(taskRepo domain.TaskRepository, interval time.Duration, maxAttempts int) *TaskWorker {
        return &TaskWorker{
                taskRepo:    taskRepo,
                handlers:    make(map[string]TaskHandler),
                interval:    interval,
                maxAttempts: maxAttempts,
                stopChan:    make(chan struct{}),
        }
}

// RegisterHandler asocia un tipo de tarea (ej: "EMAIL_SCAN", "FETCH_PROVIDER") con su handler
func (w *TaskWorker) RegisterHandler(taskType string, handler TaskHandler) {
        w.handlers[taskType] = handler
}

// Start inicia el bucle de procesamiento en segundo plano (non-blocking)
func (w *TaskWorker) Start(ctx context.Context) {
        w.wg.Add(1)
        go w.loop(ctx)
        log.Printf("[Worker] Background Task Worker iniciado (Sondeo: %v, Max Reintentos: %d)", w.interval, w.maxAttempts)
}

// Stop realiza un Graceful Shutdown del worker esperando que terminen las tareas activas
func (w *TaskWorker) Stop() {
        close(w.stopChan)
        w.wg.Wait()
        log.Println("[Worker] Background Task Worker detenido correctamente.")
}

func (w *TaskWorker) loop(ctx context.Context) {
        defer w.wg.Done()
        ticker := time.NewTicker(w.interval)
        defer ticker.Stop()

        for {
                select {
                case <-ctx.Done():
                        log.Println("[Worker] Contexto cancelado, deteniendo worker...")
                        return
                case <-w.stopChan:
                        log.Println("[Worker] Señal de parada recibida, cerrando worker...")
                        return
                case <-ticker.C:
                        w.processNextTask(ctx)
                }
        }
}

func (w *TaskWorker) processNextTask(ctx context.Context) {
        // 1. Reclamar la siguiente tarea PENDING usando FOR UPDATE SKIP LOCKED
        task, err := w.taskRepo.ClaimNextPending(ctx)
        if err != nil {
                log.Printf("[Worker] Error al consultar Task Table: %v", err)
                return
        }

        if task == nil {
                // No hay tareas pendientes en este ciclo de polling
                return
        }

        log.Printf("[Worker] Tarea tomada - ID: %s | Tipo: %s | Intento: %d/%d", task.ID, task.Type, task.Attempts, w.maxAttempts)

        // 2. Buscar el handler correspondiente al tipo de tarea
        handler, exists := w.handlers[task.Type]
        if !exists {
                errMsg := fmt.Sprintf("no existe un handler registrado para el tipo de tarea '%s'", task.Type)
                log.Printf("[Worker] Fallo en tarea %s: %s", task.ID, errMsg)
                w.handleFailure(ctx, task, errors.New(errMsg))
                return
        }

        // 3. Ejecución de la lógica de negocio asíncrona
        execErr := handler(ctx, task.Payload)
        if execErr != nil {
                log.Printf("[Worker] Falló la ejecución de la tarea %s: %v", task.ID, execErr)
                w.handleFailure(ctx, task, execErr)
                return
        }

        // 4. Éxito: Pasar estado a DONE
        if err := w.taskRepo.MarkDone(ctx, task.ID); err != nil {
                log.Printf("[Worker] Error al actualizar estado a DONE para tarea %s: %v", task.ID, err)
                return
        }

        log.Printf("[Worker] Tarea %s procesada exitosamente (DONE)", task.ID)
}

func (w *TaskWorker) handleFailure(ctx context.Context, task *domain.Task, execErr error) {
        if task.Attempts >= w.maxAttempts {
                // Superó el límite de reintentos -> FAILED definitivo
                log.Printf("[Worker] Tarea %s excedió los reintentos máximos (%d). Marcando como FAILED.", task.ID, w.maxAttempts)
                if err := w.taskRepo.MarkFailed(ctx, task.ID, execErr.Error()); err != nil {
                        log.Printf("[Worker] Error marcando tarea %s como FAILED: %v", task.ID, err)
                }
        } else {
                // Aún tiene reintentos disponibles -> Vuelve a PENDING
                log.Printf("[Worker] Reprogramando tarea %s a PENDING para reintento posterior", task.ID)
                if err := w.taskRepo.MarkRetry(ctx, task.ID); err != nil {
                        log.Printf("[Worker] Error devolviendo tarea %s a PENDING: %v", task.ID, err)
                }
        }
}
